cmake_minimum_required(VERSION 3.16)
project(CodonTraceSwarm VERSION 0.11.0 LANGUAGES CXX)
add_library(codontrace_swarm src/CodonTraceSwarm.cpp src/cts/Json.cpp src/cts/Command.cpp src/cts/Dispatcher.cpp src/cts/CommandSession.cpp src/cts/Clock.cpp src/cts/TimedTelemetry.cpp src/cts/NetworkInbox.cpp)
target_include_directories(codontrace_swarm PUBLIC src)
target_compile_features(codontrace_swarm PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(codontrace_swarm PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(CTS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(CTS_SANITIZE)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "CTS_SANITIZE requires GCC or Clang")
 endif()
 target_compile_options(codontrace_swarm PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
 target_link_options(codontrace_swarm PUBLIC -fsanitize=address,undefined)
endif()
