cmake_minimum_required(VERSION 3.16)
project(CodonTraceSwarm VERSION 0.4.0 LANGUAGES CXX)
add_library(codontrace_swarm src/CodonTraceSwarm.cpp src/cts/Json.cpp src/cts/Command.cpp src/cts/Dispatcher.cpp src/cts/CommandSession.cpp)
target_include_directories(codontrace_swarm PUBLIC src)
target_compile_features(codontrace_swarm PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(codontrace_swarm PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(CTS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(CTS_SANITIZE)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "CTS_SANITIZE requires GCC or Clang")
 endif()
 target_compile_options(codontrace_swarm PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(codontrace_swarm PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(cts_tests tests/native.cpp)
 target_link_libraries(cts_tests PRIVATE codontrace_swarm)
 add_test(NAME native COMMAND cts_tests)
 add_executable(cts_command_tests tests/commands.cpp)
 target_link_libraries(cts_command_tests PRIVATE codontrace_swarm)
 add_test(NAME commands COMMAND cts_command_tests)
 add_executable(cts_dispatcher_tests tests/dispatcher.cpp)
 target_link_libraries(cts_dispatcher_tests PRIVATE codontrace_swarm)
 add_test(NAME dispatcher COMMAND cts_dispatcher_tests)
 add_executable(cts_session_tests tests/session.cpp)
 target_link_libraries(cts_session_tests PRIVATE codontrace_swarm)
 add_test(NAME session COMMAND cts_session_tests)
 add_executable(cts_session_probe tests/session_probe.cpp)
 target_link_libraries(cts_session_probe PRIVATE codontrace_swarm)
 add_executable(cts_command_probe tests/command_probe.cpp)
 target_link_libraries(cts_command_probe PRIVATE codontrace_swarm)
endif()
